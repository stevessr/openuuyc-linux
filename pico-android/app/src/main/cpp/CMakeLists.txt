cmake_minimum_required(VERSION 3.22)
project(openuuyc_pico LANGUAGES C CXX)
find_package(OpenXR REQUIRED CONFIG)
find_library(ANDROID_LIB android REQUIRED)
find_library(LOG_LIB log REQUIRED)
find_library(EGL_LIB EGL REQUIRED)
find_library(GLES_LIB GLESv3 REQUIRED)
# Offline builds may point at a verified local Opus 1.6.1 source tree.
set(OPENUUYC_OPUS_SOURCE "" CACHE PATH "Local Opus source override")
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(OPENUUYC_OPUS_SOURCE)
  add_subdirectory("${OPENUUYC_OPUS_SOURCE}" opus-build EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(opus URL https://codeload.github.com/xiph/opus/tar.gz/refs/tags/v1.6.1
    URL_HASH SHA256=bf0b97ec7a65890b8db90ef94c4d6c18de12584c3085031953a10986f5917745)
  FetchContent_MakeAvailable(opus)
endif()
find_library(AAUDIO_LIB aaudio REQUIRED)
add_library(openuuyc_pico SHARED xr_client.cpp native_audio.cpp)
target_compile_definitions(openuuyc_pico PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_link_libraries(openuuyc_pico PRIVATE OpenXR::openxr_loader ${ANDROID_LIB} ${LOG_LIB} ${EGL_LIB} ${GLES_LIB} ${AAUDIO_LIB} opus)
